\begin{frame}{minimize $f$ over a convex set $K\subset\mathbb{R}^n$}
\begin{theorem}
\begin{minipage}{0.6\linewidth}
Suppose:
\begin{itemize}
\item $K\subset\RR^n$ is convex, and
\item $f : K \to \RR$ is in $C^1$.
\end{itemize}
If $u\in K$ is a local minimizer then
  $$\boxed{\alert{\ip{\nabla f(u)}{v-u} \ge 0}} \qquad\text{for all } v\in K.$$
\end{minipage}
\hfill
\begin{minipage}{0.36\linewidth}
  \centering
  \scalebox{1.0}{\input{tikz/convexmin.tex}}
\end{minipage}
\end{theorem}

\smallskip
\emph{proof:}\, Note that \alert{$(1-t) u + t v = u+t(v-u)\in K$} for $0\le t\le 1$.

But $t=0$ locally minimizes $g(t)=f(u+t(v-u))$ on $[0,1]$.

By the calculus I problem, $0\le g'(0)=\ip{\nabla f(u)}{v-u}$. \qed

\medskip
\grey{idea: $\grad f(u)\ne 0$ is possible, but $f$ does not decrease as we move from $u$ into $K$}
\end{frame}


\begin{frame}{minimize convex $f$ over a convex set $K\subset\mathbb{R}^n$}

\begin{theorem}[narrowed to a characterization]
\begin{minipage}{0.6\linewidth}
Suppose:
\begin{itemize}
\item $K\subset\RR^n$ is convex, and
\item $f : K \to \RR$ is in $C^1$ \alert{and convex}.
\end{itemize}
Then $u\in K$ is a minimizer \alert{if and only if}
  $$\boxed{\alert{\ip{\nabla f(u)}{v-u} \ge 0}} \qquad\text{for all } v\in K.$$
\end{minipage}
\hfill
\begin{minipage}{0.36\linewidth}
  \centering
  \scalebox{1.0}{\input{tikz/convexmin.tex}}
\end{minipage}
\end{theorem}

\begin{itemize}
\item the boxed \alert{variational inequality} is one way to state the KKT condition
\item $u$ exists if additionally $K\ne \emptyset$ and
    \begin{itemize}
    \item[$\lc$] $K$ is compact, or
    \item[$\lc$] $K$ is closed and $f$ is \emph{coercive}: $f\to \infty$ as $\|v\|\to \infty$ for $v\in K$
    \end{itemize}
\item $u$ is unique if additionally $f$ strictly convex
\end{itemize}
\end{frame}

\begin{frame}{constrained nonlinear equations}

\begin{itemize}
\item change the problem minimally: solve a variational inequality instead of ``$F(u)=0$''
\end{itemize}

\medskip
\begin{theorem}[Hartman \& Stampacchia, 1964]
If $K\subset\mathbb{R}^n$ is nonempty, compact, and convex, and if $F:K \to\RR^n$ is continuous, then there exists $u\in K$ so that
    $$\boxed{\alert{\ip{F(u)}{v-u} \ge 0}} \qquad\text{for all } v\in K.$$
\end{theorem}

\emph{proof:}\, The map $v\mapsto \Pi_K\big(v-F(v)\big)$ is continuous from $K$ to $K$.  By
Brouwer it has a fixed point:
      $$u = \Pi_K \big(u - F(u)\big).$$
Let $z = u - F(u)$ so $u=\Pi_K z$.  By the projection theorem, for all $v\in K$,
      $$0 \le \ip{u - z}{v - u} = \ip{F(u)}{v-u}. \qquad\qed$$
\end{frame}

\begin{theorem}[Hartman \& Stampacchia, 1964]
If $K\subset\mathbb{R}^n$ is nonempty, compact, and convex, and if $F:K \to\RR^n$ is continuous, then there exists $u\in K$ so that
    $$\boxed{\alert{\ip{F(u)}{v-u} \ge 0}} \qquad\text{for all } v\in K.$$
Furthermore, if $u$ is in the interior of $K$ then $F(u)=0$.
\end{theorem}

\item what about uniqueness here? sufficient hypothesis on $F$?

\pause
\bigskip
\item suppose for $u_1,u_2\in K$: \, $\ip{F(u_i)}{v-u_i} \ge 0$ for all $v\in K$
\item substitute $v=u_2$ and then $v=u_1$, then add:
	$$\ip{F(u_1) - F(u_2)}{u_1-u_2} \le 0 \hspace{10mm} \leftarrow \text{\small for uniqueness, \alert{block this} when $u_1\ne u_2$}$$

\begin{frame}{a well-posed problem}

\begin{definition}
$F:K \to \RR^n$ is \alert{monotone} if
    $$\ip{F(v)-F(w)}{v-w} \ge 0$$
for all $v,w\in K$, and \alert{strictly monotone} if equality implies $v=w$.
\end{definition}

\begin{corollary}
If additionally $F$ is strictly monotone then $u\in K$ is unique.
\end{corollary}

\begin{itemize}
\item but is this hypothesis reasonable?

\pause
\item yes, it is the same as strict convexity of the objective function
\end{itemize}

\begin{proposition}[optimization case]
Suppose $f:\RR^n \to \RR$ is $C^1$.  Then $f$ is strictly convex if and only if $F=\grad f$ is strictly monotone.
\end{proposition}
\end{frame}


\begin{frame}{variational inequalities}

\begin{definitions}[made official]
Given a normed vector space $X$, a closed and convex \alert{admissible subset} $K\subset X$, and $F: K\to X'$ continuous, but generally nonlinear, find $u\in K$ so that
  $$\boxed{\alert{\ip{F(u)}{v-u} \ge 0}} \qquad \text{for all } v\in K.$$
This is the \alert{variational inequality} problem for the data $K$ and $F$.
\end{definitions}

\bigskip
  {\centering\small
    \begin{tabular}{lll}
      example problem             & typical admissible set           & variational inequality \\
      \midrule
      calculus minimization       & $[a,b]\subset\RR$                & $f'(u)(v-u) \ge 0 \quad\,\forall v\in [a,b]$ \\
      projection of $x$ onto $K$  & $K\subset H$ closed, convex        & $\ip{u-x}{v-u} \ge 0 \quad\forall v\in K$ \\
      minimize $f$ over $K$       & $K\subset \RR^n$ compact, convex    & $\ip{\nabla f(u)}{v-u} \ge 0 \quad\forall v\in K$ \\
      constrain nonlinear eqns to $K$ & $K\subset \RR^n$ compact, convex   & $\ip{F(u)}{v-u} \ge 0 \quad\,\forall v\in K$ \\
      \bottomrule
    \end{tabular}
  \par}

\bigskip
{\small \grey{perspective: \, VI generalizes system $F(u)=0$ to convex input sets, enabling existence}}
\end{frame}

\begin{frame}{theory interruption}

\begin{definition}
A bilinear form $a(\cdot,\cdot)$ is \alert{coercive} on a Hilbert space $H$ if there is $\alpha>0$ so that
	$$a(v,v) \ge \alpha \|v\|^2 \quad \text{for all } v\in H.$$
\end{definition}

\begin{itemize}
\item by the Poincar\'e inequality the bilinear form in the obstacle problem, $a(u,v) = \int_\Omega \nabla u \cdot \nabla v$ is coercive on $H_0^1(\Omega)$
\item coercive $\implies$ strictly monotone
\end{itemize}

\begin{theorem}[Stampacchia 1964; generalizes Lax--Milgram to $K\subset H$]
Suppose $a(\cdot,\cdot)$ is bilinear, continuous, and coercive on a Hilbert space $H$.  Suppose $K \subset H$ is nonempty, closed, and convex.  Then for $\ell\in H'$ there exists a unique $u\in K$ with
    $$a(u,v-u)\ge \ip{\ell}{v-u} \quad\forall v\in K.$$
\end{theorem}

%{\small \emph{proof:}\, Apply parallelogram law and completeness argument for the symmetric part, and use the Banach fixed-point theorem incrementally to add-back the antisymmetric part.  \qed}
\end{frame}

\begin{frame}{Signorini contact problem (VI form)}

\begin{itemize}
\item define convex and closed admissible set:
    $$\alert{K = \{v\in H^1(\Omega;\mathbb{R}^d) : v=u_D \text{ on } \Gamma_D \text{ and } v\cdot n\le g \text{ on } \Gamma_C\}}$$
\item VI problem is to find \alert{$u\in K$} so that
    $$\alert{\int_\Omega \sigma(u):E(v-u) \ge \int_\Omega f\cdot(v-u) \qquad\text{for all } v\in K}$$
\item equivalent to minimizing the elastic energy over admissible displacements:
    $$u \leftarrow \min_{v\in K} \int_\Omega \tfrac12\sigma(v):E(v) - f\cdot v$$

\pause
\begin{theorem}[Fichera 1964]
The Signorini problem is well-posed.
\end{theorem}

\item key idea: \, $a(v,w)=\int_\Omega \sigma(v):E(w)$ is coercive on $H_0^1(\Omega;\mathbb{R}^d)$ by Korn's inequality
\end{itemize}
\end{frame}

u \leftarrow \min_{v\in K} J(v) = \int_\Omega W(I+\nabla v) - f\cdot v

\smallskip
    \item[$\lc$] $\bq(x,u,\grad u)$ is the vertically-summed \alert{flux},

e.g.~gravity-driven or wind/ocean-driven

\begin{frame}{fluid layer in a climate: implicit time step}

\begin{itemize}
\item consider backward Euler time steps \hfill \pcomm{parabolic form open?}
\item for $X=W^{1,p}(\Omega)$ let \alert{$K=\{v\in X : v\ge 0\}$} be the \alert{admissible thicknesses}
\item given previous (time $t_{n-1}$) thickness $u_{n-1}\in K$
\item VI problem is to find $u_n\in K$ at $t_n$ so that, for all $v\in K$,
    $$\alert{\int_\Omega \frac{u_n-u_{n-1}}{\Delta t}(v-u_n) - \bq_n\cdot\nabla(v-u_n) \ge \int_\Omega f_n\,(v-u_n)} \hspace{20mm}$$
\item \alert{$\bq_n = \bq(x,u_n,\grad u_n)$} denotes the unknown flux
\item well-posed for many flux formulas \pcomm{B.~2021}
    \begin{itemize}
    \item[$\lc$] advective, diffusive, nonlinear, \dots
    \end{itemize}
\item generally not an optimization
\item note: $\Delta t\to \infty$ limit is steady state
\end{itemize}

\vspace{-16mm}

\end{frame}


\begin{frame}{fluid layer in a climate: implicit time step}

VI model for updated thickness $\alert{u_n\in K = \{v\in X : v\ge 0\}}$:
    $$\alert{\int_\Omega \frac{u_n-u_{n-1}}{\Delta t}(v-u_n) - \bq_n\cdot\nabla(v-u_n) \ge \int_\Omega f_n\,(v-u_n)} \hspace{35mm}$$

\begin{columns}[T,onlytextwidth]
\begin{column}{0.6\textwidth}
\begin{itemize}
\item solution decomposes $\Omega$ into disjoint sets:
  \begin{align*}
    \Omega_n      &= \{u_n>0\}                        && \grey{\text{\small fluid-covered area}} \\
    \Omega_n^r    &= \{u_n=0 \text{ and } u_{n-1}>0\} && \grey{\text{\small retreat area}} \\
    \Omega_n^{00} &= \{u_n=0 \text{ and } u_{n-1}=0\} && \grey{\text{\small remaining uncovered}}
  \end{align*}
\item note that $u_{n-1}+\Delta t\,f_n\le 0$ on $\Omega_n^r\cup\Omega_n^{00}$
\end{itemize}
\end{column}

\begin{frame}{FE for VIs: Falk's lemma}

\begin{itemize}
\item for elliptic PDEs, the basic FE accuracy story starts with Cea's lemma
    \begin{itemize}
    \item[$\lc$] the \alert{error} is $u-u_h$
    \end{itemize}
\item what error norm can we expect for a VI like the classical obstacle problem?

\pause
\medskip
\item answer: Falk's 1974 generalization of C\'ea's lemma \pcomm{next slide}
\item we state the lemma using an abstracted set-up:

\smallskip
    \begin{itemize}
    \item[$\lc$] assume $K_h\subset X_h \subset X$ \, \pcomm{but generally $K_h \not\subset K$}

\smallskip
    \item[$\lc$] suppose continuity for the bilinear form: $a(v,w)\le M\|v\|\|w\|$ for all $v,w\in X$

\smallskip
    \item[$\lc$] and coercivity: $a(v,v)\ge \alpha \|v\|^2$ for all $v\in X$

\smallskip
    \item[$\lc$] define a linear operator $A:X\to X'$, so that $a(v,w)=\ip{Av}{w}$ for all $v,w\in X$

\smallskip
    \item the \alert{residual} of $w\in X$ is \alert{$F(w)=Aw-\ell \in X'$}, for a source term $\ell \in X'$
    \end{itemize}
\end{itemize}
\end{frame}


\begin{frame}{FE for VIs: Falk's lemma}

\begin{itemize}
\item continuum problem for $u\in K$: \, $a(u,v-u) \ge \ip{\ell}{v-u} \,\forall v\in K$
    \begin{itemize}
    \item[$\lc$] for the exact solution $u\in K$, generally \alert{$F(u) = Au-\ell \ne 0$}
        \end{itemize}

\vspace{-12mm}
\hfill \input{tikz/solnsupport.tex}

\vspace{-8mm}
\item FE problem for $u_h\in K_h$: \, $a(u_h,v_h-u_h) \ge \ip{\ell}{v_h-u_h} \,\forall v_h\in K_h$
\end{itemize}

\begin{lemma}[Falk, 1974]

\vspace{-5mm}
\begin{align*}
\|u-u_h\|_X^2 &\le \frac{M^2}{\alpha^2} \inf_{v_h\in K_h} \|u - v_h\|^2 \\
   &\qquad \alert{+ \frac{2}{\alpha} \left(\inf_{v\in K} \ip{F(u)}{v-u_h} + \inf_{v_h\in K_h} \ip{F(u)}{v_h-u}\right)}
\end{align*}
\end{lemma}

\begin{overlayarea}{\textwidth}{12mm}
\vspace{-2mm}
\only<1>{
\begin{corollary}[C\'ea's lemma]
$K=X$: \qquad $\|u-u_h\|_X \le \frac{M}{\alpha} \inf_{v_h\in X_h} \|u - v_h\|$.
\end{corollary}
}
\only<2>{
\begin{corollary}[more conforming case]
$K_h \subset K$: \qquad ``$\inf_{v\in K}$'' term can be replaced with zero
\end{corollary}
}
\only<3>{
\begin{corollary}[first-order convergence]
assuming quasi-uniform (regular) meshes: \qquad $\|u-u_h\|_X \le C h$
\end{corollary}
}
\end{overlayarea}
