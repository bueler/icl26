\begin{frame}{minimize $f$ over a convex set $K\subset\mathbb{R}^n$}
\begin{theorem}
\begin{minipage}{0.6\linewidth}
Suppose:
\begin{itemize}
\item $K\subset\RR^n$ is convex, and
\item $f : K \to \RR$ is in $C^1$.
\end{itemize}
Then a local minimizer $u\in K$ satisfies
  $$\boxed{\alert{\ip{\nabla f(u)}{v-u} \ge 0}} \qquad\text{for all } v\in K.$$
\end{minipage}
\hfill
\begin{minipage}{0.36\linewidth}
  \centering
  \scalebox{1.0}{\input{tikz/convexmin.tex}}
\end{minipage}
\end{theorem}

\pause
\medskip
\emph{proof:}\, If $v\in K$ then $(1-t) u + t v = u+t(v-u)\in K$ for $0\le t\le 1$.

We know that $t=0$ locally minimizes $g(t)=f(u+t(v-u))$ on $[0,1]$.

By the calculus I problem, $0\le g'(0)=\ip{\nabla f(u)}{v-u}$. \qed

\medskip
\grey{idea: $\grad f(u)\ne 0$ is possible, but $f$ does not decrease as we start from $u$ and head into $K$}
\end{frame}


\begin{frame}{minimize convex $f$ over a convex set $K\subset\mathbb{R}^n$}

\begin{theorem}[narrowed to a characterization]
\begin{minipage}{0.6\linewidth}
Suppose:
\begin{itemize}
\item $K\subset\RR^n$ is convex, and
\item $f : K \to \RR$ is in $C^1$ \alert{and convex}.
\end{itemize}
Then $u\in K$ is a minimizer \alert{if and only if}
  $$\boxed{\alert{\ip{\nabla f(u)}{v-u} \ge 0}} \qquad\text{for all } v\in K.$$
\end{minipage}
\hfill
\begin{minipage}{0.36\linewidth}
  \centering
  \scalebox{1.0}{\input{tikz/convexmin.tex}}
\end{minipage}
\end{theorem}

\begin{itemize}
\item $u$ exists if $K\ne \emptyset$ and
    \begin{itemize}
    \item[$\lc$] $K$ is compact, or
    \item[$\lc$] $K$ is closed and $f$ is \emph{coercive}: $f\to \infty$ as $\|v\|\to \infty$ for $v\in K$
    \end{itemize}
\item $u$ is unique if $f$ strictly convex
\item the boxed variational inequality is one way to state the KKT conclusion
\end{itemize}

\begin{frame}{constrained nonlinear equations}

\begin{itemize}
\item change the problem minimally: solve a variational inequality instead of ``$F(u)=0$''
\end{itemize}

\medskip
\begin{theorem}[Hartman \& Stampacchia, 1964]
If $K\subset\mathbb{R}^n$ is nonempty, compact, and convex, and if $F:K \to\RR^n$ is continuous, then there exists $u\in K$ so that
    $$\boxed{\alert{\ip{F(u)}{v-u} \ge 0}} \qquad\text{for all } v\in K.$$
\end{theorem}

\emph{proof:}\, The map $v\mapsto \Pi_K\big(v-F(v)\big)$ is continuous from $K$ to $K$.  By
Brouwer it has a fixed point:
      $$u = \Pi_K \big(u - F(u)\big).$$
Let $z = u - F(u)$ so $u=\Pi_K z$.  By the projection theorem, for all $v\in K$,
      $$0 \le \ip{u - z}{v - u} = \ip{F(u)}{v-u}. \qquad\qed$$
\end{frame}

\begin{frame}{a well-posed problem}

\begin{definition}
$F$ is \alert{monotone} on $K$ if
    $$\ip{F(v)-F(w)}{v-w} \ge 0$$
for all $v,w\in K$, and \alert{strictly monotone} if equality implies $v=w$.
\end{definition}

\begin{corollary}
If additionally $F$ is strictly monotone then $u\in K$ is unique.
\end{corollary}

\pause
\begin{itemize}
\item is this hypothesis reasonable?
\item in optimization case, it matches strict convexity of the objective function
\end{itemize}

\begin{proposition}
Suppose $f:\RR^n \to \RR$ is $C^1$.  Then $f$ is strictly convex if and only if $F=\grad f$ is strictly monotone.
\end{proposition}
\end{frame}


\begin{frame}{variational inequalities}

\begin{definitions}[made official!]
Given a normed vector space $X$, a closed and convex \alert{admissible subset} $K\subset X$, and $F: K\to X'$ continuous, but generally nonlinear, find $u\in K$ so that
  $$\boxed{\alert{\ip{F(u)}{v-u} \ge 0}} \qquad \text{for all } v\in K.$$
This is the \alert{variational inequality} problem for the data $K$ and $F$, denoted $\operatorname{VI}(K,F)$.
\end{definitions}

\bigskip
  {\centering\small
    \begin{tabular}{lll}
      example                     & admissible set           & variational inequality \\
      \midrule
      calculus I min problem      & $[a,b]\subset\RR$                & $f'(u)(x-u) \ge 0 \quad\forall x\in [a,b]$ \\
      projection of $x$ onto $K$  & $K\subset H$ closed, convex        & $\ip{u-x}{y-u} \ge 0 \quad\forall y\in K$ \\
      minimize $f$ over $K$       & $K\subset \RR^n$ closed, convex    & $\ip{\nabla f(u)}{v-u} \ge 0 \quad\forall v\in K$ \\
      constrained nonlinear eqns  & $K\subset \RR^n$ compact, convex   & $\ip{F(u)}{v-u} \ge 0 \quad\forall v\in K$ \\
      \bottomrule
    \end{tabular}
  \par}

\bigskip
{\small \grey{perspective: \, VIs generalize equations $F(u)=0$ to convex input sets, enabling existence}}
\end{frame}

\begin{frame}{classical obstacle problem (weak forms)}

\begin{itemize}
\item for theoretical and finite-element purposes, we want to rewrite in \alert{weak form}
    \begin{itemize}
    \item[$\lc$] in applications, this step is not as obvious as for the equality-constrained case?
    \end{itemize}
\item let \alert{$K = \{v\in H^1(\Omega) : v\ge\psi \text{ and } v|_{\partial\Omega}=g\}$}, a closed, convex subset \,\grey{(not a subspace)}
\end{itemize}

\begin{block}{weak forms}
the classical obstacle problem can be stated as

\begin{columns}[T]
\begin{column}{0.45\textwidth}
\begin{itemize}
\item a \alert{minimization problem}:
    $$u \leftarrow \min_{v\in K} \int_\Omega \tfrac12|\nabla v|^2 - fv$$
\end{itemize}
\end{column}
\begin{column}{0.55\textwidth}
\begin{itemize}
\item a \alert{variational inequality} (VI):
    $$\int_\Omega \nabla u\cdot\nabla(v-u) \ge \int_\Omega f\,(v-u) \quad \forall v\in K$$
\end{itemize}
\end{column}
\end{columns}
\end{block}

\begin{proposition}
\begin{itemize}
\item the problems are equivalent, and well-posed
\item if $u\in K$ is regular enough to solve the complementarity problem (strong form) then it solves these weak forms
\end{itemize}
\end{proposition}
\end{frame}


\begin{frame}{theory interruption}

\begin{definition}
A bilinear form $a(\cdot,\cdot)$ is \alert{coercive} if there is $\alpha>0$ so that
	$$a(v,v) \ge \alpha \|v\|^2 \quad \text{for all } v\in H.$$
\end{definition}

\begin{itemize}
\item the bilinear form in the obstacle problem, $a(u,v) = \int_\Omega \nabla u \cdot \nabla v$, is coercive on $H_0^1(\Omega)$, by the Poincar\'e inequality
\item coercive $\implies$ strictly monotone
\end{itemize}

\begin{theorem}[Stampacchia 1964; generalizes Lax--Milgram (1954) to $K\subset H$]
Suppose $a(\cdot,\cdot)$ is bilinear, bounded, and coercive on a Hilbert space $H$, and $K \subset H$ is nonempty, closed, and convex.  Then for each $\ell\in H'$ there is a unique $u\in K$ with
    $$a(u,v-u)\ge \ip{\ell}{v-u} \quad\forall v\in K.$$
\end{theorem}

%{\small \emph{proof:}\, Apply parallelogram law and completeness argument for the symmetric part, and use the Banach fixed-point theorem incrementally to add-back the antisymmetric part.  \qed}
\end{frame}

\begin{frame}{Signorini contact problem (VI form)}

\begin{itemize}
\item define convex and closed admissible set:
    $$\alert{K = \{v\in H^1(\Omega;\mathbb{R}^d) : v=u_D \text{ on } \Gamma_D \text{ and } v\cdot n\le g \text{ on } \Gamma_C\}}$$
\item VI problem is to find \alert{$u\in K$} so that
    $$\alert{\int_\Omega \sigma(u):E(v-u) \ge \int_\Omega f\cdot(v-u) \qquad\text{for all } v\in K}$$

\pause
\item key: bilinear form $a(v,w)=\int_\Omega \sigma(v):E(w)$ is coercive on $H_0^1(\Omega;\mathbb{R}^d)$ by Korn's inequality

\begin{theorem}[Fichera 1964]
The Signorini problem is well-posed; there exists a unique solution.
\end{theorem}

\item equivalent to minimizing the elastic energy over admissible displacements:
    $$u \leftarrow \min_{v\in K} \int_\Omega \tfrac12\sigma(v):E(v) - f\cdot v$$
\end{itemize}
\end{frame}

\item hyperelastic: stored energy density $W(F)$, e.g.~compressible neo-Hookean
      $$
        W(F) = \tfrac{\mu}{2}\left(|F|^2-2\right) - \mu\ln J + \tfrac{\lambda}{2}(\ln J)^2
      $$
\item minimize $\displaystyle \Pi(v) = \int_\Omega W(I+\nabla v) - B\cdot v$
      \; over the same kind of $\cK$
\item a minimizer solves the VI
      $$
        \ip{\Pi'(u)}{v-u}\ge 0 \qquad\text{for all } v\in\cK
      $$
\item linearizing $W$ at $F=I$ recovers the Signorini problem
\item but $W$ is \alert{not convex} \grey{(Ciarlet 2022)}, so the VI is only a
      \emph{necessary} condition, and solutions need not be unique
\end{itemize}
\end{frame}

\begin{frame}{fluid layer: one implicit time step}
\begin{columns}[T]
\begin{column}{0.6\textwidth}
  given $u_{n-1}\ge0$, find $u_n\ge0$ so that
  $$
    \frac{u_n-u_{n-1}}{\Delta t}+\nabla\cdot Q_n = F_n \quad \text{where } u_n>0
  $$
  this decomposes $\Omega$ into disjoint sets:
  \begin{align*}
    \Omega_n         &= \{u_n>0\} \\
    \Omega_n^r       &= \{u_n=0 \text{ and } u_{n-1}>0\} \quad \grey{\text{retreat}} \\
    \Omega_n^{00}    &= \{u_n=0 \text{ and } u_{n-1}=0\}
  \end{align*}
  \vspace{-2mm}
  \uncover<2->{
  VI on $\cK=\{v\in W^{1,p}(\Omega) : v\ge 0\}$: for all $v\in\cK$,
  $$
    \int_\Omega \frac{u_n-u_{n-1}}{\Delta t}(v-u_n) - Q_n\cdot\nabla(v-u_n)
    \ge \int_\Omega F_n\,(v-u_n)
  $$}
\end{column}
\begin{column}{0.38\textwidth}
  \centering
  \includegraphics[width=0.85\textwidth]{figs/bueler2021-fig3.png}

  \medskip
  \raggedright
  {\small on $\Omega_n^r\cup\Omega_n^{00}$: \, $u_{n-1}+\Delta t\,F_n\le 0$}

  \smallskip
  {\small \grey{mass is \emph{not} exactly conserved on the retreat set}}
\end{column}
\end{columns}
\end{frame}

  \grey{Falk, Math.\ Comp.\ 1974}
